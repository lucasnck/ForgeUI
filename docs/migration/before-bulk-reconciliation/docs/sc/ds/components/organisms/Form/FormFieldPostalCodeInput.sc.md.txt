## Hooks

```ts
"use client";
import { useId } from "react";

import { useFormContext, useController } from "react-hook-form";
import type { FieldPathByValue, FieldValues } from "react-hook-form";
import type { FormFieldPostalCodeInputProps } from "./FormFieldPostalCodeInput";

export function useFormFieldPostalCodeInput<
  TValues extends FieldValues,
  TName extends FieldPathByValue<TValues, string> = FieldPathByValue<TValues, string>,
>({
  id,
  name,
  label,
  hint,
  error: displayedError,
  required,
  disabled: callerDisabled,
  ...inputProps
}: FormFieldPostalCodeInputProps<TValues, TName>) {
  const generatedId = useId();
  const methods = useFormContext<TValues>();
  if (!methods) throw new Error("FormFieldPostalCodeInput requires FormProvider.");
  const disabled = callerDisabled || methods.formState.disabled;
  const { field, fieldState } = useController({ control: methods.control, name, disabled });
  const value: unknown = field.value;
  if (typeof value !== "string")
    throw new Error("FormFieldPostalCodeInput requires string defaults.");
  const error = fieldState.error;
  const message =
    displayedError ?? (typeof error?.message === "string" ? error.message : undefined);

  return {
    controlId: id ?? generatedId,
    label,
    hint,
    required,
    disabled,
    message,
    invalid: Boolean(error) || Boolean(displayedError),
    inputProps,
    binding: {
      name: field.name,
      ref: field.ref,
      onBlur: field.onBlur,
      value,
      onValueChange: field.onChange,
    },
  };
}
```

## Content

```tsx
"use client";
import type { FieldPathByValue, FieldValues } from "react-hook-form";
import type {
  FormFieldPostalCodeInputControlProps,
  FormFieldRegistrationProps,
} from "../../../../types";

export interface FormFieldPostalCodeInputProps<
  TValues extends FieldValues,
  TName extends FieldPathByValue<TValues, string> = FieldPathByValue<TValues, string>,
>
  extends FormFieldPostalCodeInputControlProps, FormFieldRegistrationProps<TValues, TName> {}

import { PostalCodeInput } from "../../../molecules/PostalCodeInput";

import { FormField } from "../../../molecules/FormField";
import { useFormFieldPostalCodeInput } from "./useFormFieldPostalCodeInput";

export function FormFieldPostalCodeInput<
  TValues extends FieldValues,
  TName extends FieldPathByValue<TValues, string> = FieldPathByValue<TValues, string>,
>(props: FormFieldPostalCodeInputProps<TValues, TName>) {
  const view = useFormFieldPostalCodeInput(props);

  return (
    <FormField controlId={view.controlId} label={view.label} hint={view.hint} error={view.message}>
      <PostalCodeInput
        {...view.inputProps}
        {...view.binding}
        id={view.controlId}
        disabled={view.disabled}
        invalid={view.invalid}
        aria-required={view.required || undefined}
      />
    </FormField>
  );
}
```
