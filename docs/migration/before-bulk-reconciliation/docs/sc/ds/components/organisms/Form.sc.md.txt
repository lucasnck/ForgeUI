## Hooks

```tsx
"use client";

import type { SubmitEvent } from "react";
import type { FormProps } from "./Form";

export function useForm({ onValidSubmit, onSubmit, loading = false, loadingLabel, children, className, ...props }: FormProps) {
  function submit(event: SubmitEvent<HTMLFormElement>) {
    onSubmit?.(event);
    if (event.defaultPrevented) return;
    event.preventDefault();
    if (loading) return;
    const form = event.currentTarget;
    const environment = form.ownerDocument.defaultView;
    if (!environment) return;
    const nativeEvent = event.nativeEvent;
    const submitter = nativeEvent instanceof environment.SubmitEvent ? nativeEvent.submitter : undefined;
    onValidSubmit(new environment.FormData(form, submitter ?? undefined), event);
  }
  return { context: { loading, loadingLabel }, children, className, props, submit };
}
```

## Content

```tsx
"use client";

import type { SubmitEvent } from "react";
import type { NativeFormRootProps } from "../../../types/collection-contracts";
import type { ComponentChildrenProps } from "../../../types/compound-composition";
import type { ComponentLoadingProps } from "../../../types/component-capabilities";
import { FormContext } from "../../../internal/form/FormContext";
import { collectCompoundParts } from "../../../helpers/compound";
import { FormHeader } from "../../atoms/FormHeader";
import { FormBody } from "../../atoms/FormBody";
import { FormFooter } from "../../atoms/FormFooter";
import { useForm } from "./useForm";
import styles from "./Form.module.css";

export interface FormProps extends NativeFormRootProps, ComponentChildrenProps, ComponentLoadingProps {
  onValidSubmit(data: FormData, event: SubmitEvent<HTMLFormElement>): void;
}

export function Form(input: FormProps) {
  const { context, children, props, className, submit } = useForm(input);
  const parts = collectCompoundParts(children, { header: FormHeader, body: FormBody, footer: FormFooter });
  return (
    <FormContext value={context}>
      <form
        {...props}
        onSubmit={submit}
        className={[styles.root, className].filter(Boolean).join(" ")}
        data-gap="2"
      >
        {parts.header}
        {parts.body}
        {parts.footer}
      </form>
    </FormContext>
  );
}
```

```tsx
"use client";

import { createContext, useContext } from "react";

export interface FormContextValue {
  loading: boolean;
  loadingLabel?: string;
}
export const FormContext = createContext<FormContextValue | null>(null);
export function useFormPresentationContext() {
  const context = useContext(FormContext);
  if (!context) throw new Error("Form parts require Form");
  return context;
}
```
