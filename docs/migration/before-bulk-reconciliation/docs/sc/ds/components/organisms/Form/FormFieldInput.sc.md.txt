## Hooks

```ts
"use client";

import { useId } from "react";
import { useFormContext, useFormState } from "react-hook-form";
import type { FieldPath, FieldValues } from "react-hook-form";
import type { FormFieldInputProps } from "./FormFieldInput";

export function useFormFieldInput<
  TValues extends FieldValues,
  TName extends FieldPath<TValues> = FieldPath<TValues>,
>({
  id,
  name,
  disabled,
  label,
  hint,
  required = false,
  inputType = "text",
  error: displayedError,
  ...inputProps
}: FormFieldInputProps<TValues, TName>) {
  const generatedId = useId();
  const controlId = id ?? generatedId;
  const methods = useFormContext<TValues>();
  if (!methods) throw new Error("FormFieldInput requires FormProvider.");
  const state = useFormState({ control: methods.control, name, exact: true });
  const { error } = methods.getFieldState(name, state);
  const message =
    displayedError ?? (typeof error?.message === "string" ? error.message : undefined);
  const registration = methods.register(name, { disabled: disabled || methods.formState.disabled });

  return {
    inputType,
    controlId,
    label,
    hint,
    required,
    disabled: disabled || methods.formState.disabled,
    inputProps,
    registration,
    error,
    message,
  };
}
```

## Content

```tsx
"use client";

import type { FormTextInputType } from "@/lib/forms";
import type { FieldPath, FieldValues } from "react-hook-form";
import type {
  ComponentFullWidthProps,
  FormFieldRegistrationProps,
  NativeFormFieldInputProps,
  VisualProps,
} from "../../../../types";
import { FormField } from "../../../molecules/FormField";
import { Input } from "../../../atoms/Input";
import { useFormFieldInput } from "./useFormFieldInput";
import styles from "./FormFieldInput.module.css";

export interface FormFieldInputProps<
  TValues extends FieldValues,
  TName extends FieldPath<TValues> = FieldPath<TValues>,
>
  extends
    NativeFormFieldInputProps,
    VisualProps,
    ComponentFullWidthProps,
    FormFieldRegistrationProps<TValues, TName> {
  inputType?: FormTextInputType;
}

export function FormFieldInput<
  TValues extends FieldValues,
  TName extends FieldPath<TValues> = FieldPath<TValues>,
>(props: FormFieldInputProps<TValues, TName>) {
  const view = useFormFieldInput(props);

  return (
    <div className={styles.root}>
      <FormField
        controlId={view.controlId}
        label={view.label}
        hint={view.hint}
        error={view.message}
      >
        <Input
          {...view.inputProps}
          {...view.registration}
          type={view.inputType}
          disabled={view.disabled}
          invalid={Boolean(view.error)}
          aria-required={view.required || undefined}
        />
      </FormField>
    </div>
  );
}
```
