## Hooks

```ts
"use client";
import { useId } from "react";

import { useFormContext, useFormState } from "react-hook-form";
import type { FieldPathByValue, FieldValues } from "react-hook-form";
import type { FormFieldTelephoneInputProps } from "./FormFieldTelephoneInput";

export function useFormFieldTelephoneInput<
  TValues extends FieldValues,
  TName extends FieldPathByValue<TValues, string> = FieldPathByValue<TValues, string>,
>({
  id,
  name,
  label,
  hint,
  error: displayedError,
  required,
  disabled: callerDisabled,
  ...inputProps
}: FormFieldTelephoneInputProps<TValues, TName>) {
  const generatedId = useId();
  const methods = useFormContext<TValues>();
  if (!methods) throw new Error("FormFieldTelephoneInput requires FormProvider.");
  const disabled = callerDisabled || methods.formState.disabled;
  const state = useFormState({ control: methods.control, name, exact: true });
  const { error } = methods.getFieldState(name, state);
  const registration = methods.register(name, { disabled });
  const defaultValue: unknown = methods.getValues(name);
  if (typeof defaultValue !== "string")
    throw new Error("Native text fields require string defaults.");
  const message =
    displayedError ?? (typeof error?.message === "string" ? error.message : undefined);

  return {
    controlId: id ?? generatedId,
    label,
    hint,
    required,
    disabled,
    message,
    invalid: Boolean(error) || Boolean(displayedError),
    inputProps,
    binding: { ...registration, defaultValue },
  };
}
```

## Content

```tsx
"use client";
import type { FieldPathByValue, FieldValues } from "react-hook-form";
import type {
  FormFieldTelephoneInputControlProps,
  FormFieldRegistrationProps,
} from "../../../../types";

export interface FormFieldTelephoneInputProps<
  TValues extends FieldValues,
  TName extends FieldPathByValue<TValues, string> = FieldPathByValue<TValues, string>,
>
  extends FormFieldTelephoneInputControlProps, FormFieldRegistrationProps<TValues, TName> {}

import { TelephoneInput } from "../../../molecules/TelephoneInput";

import { FormField } from "../../../molecules/FormField";
import { useFormFieldTelephoneInput } from "./useFormFieldTelephoneInput";

export function FormFieldTelephoneInput<
  TValues extends FieldValues,
  TName extends FieldPathByValue<TValues, string> = FieldPathByValue<TValues, string>,
>(props: FormFieldTelephoneInputProps<TValues, TName>) {
  const view = useFormFieldTelephoneInput(props);

  return (
    <FormField controlId={view.controlId} label={view.label} hint={view.hint} error={view.message}>
      <TelephoneInput
        {...view.inputProps}
        {...view.binding}
        id={view.controlId}
        disabled={view.disabled}
        invalid={view.invalid}
        aria-required={view.required || undefined}
      />
    </FormField>
  );
}
```
