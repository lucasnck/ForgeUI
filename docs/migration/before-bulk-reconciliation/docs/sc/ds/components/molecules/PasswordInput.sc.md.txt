## Hooks

```tsx
"use client";

import { useImperativeHandle, useLayoutEffect, useRef, useState } from "react";
import type { PasswordInputProps } from "./PasswordInput";

export function usePasswordInput({ revealable = true, disabled = false, concealLabel, revealLabel, fullWidth = true, ref, ...props }: PasswordInputProps) {
  const [revealed, setRevealed] = useState(false);
  const [previousRevealable, setPreviousRevealable] = useState(revealable);
  if (previousRevealable !== revealable) {
    setPreviousRevealable(revealable);
    if (!revealable) setRevealed(false);
  }
  const inputRef = useRef<HTMLInputElement>(null);
  const selection = useRef<{ start: number | null; end: number | null; direction: HTMLInputElement["selectionDirection"] } | null>(null);
  useImperativeHandle(ref, () => inputRef.current!, []);
  const showText = revealable && revealed;
  function toggle() {
    const input = inputRef.current;
    if (disabled || !revealable) return;
    if (input) selection.current = { start: input.selectionStart, end: input.selectionEnd, direction: input.selectionDirection };
    setRevealed(current => !current);
  }
  useLayoutEffect(() => {
    const saved = selection.current;
    const input = inputRef.current;
    selection.current = null;
    if (input && saved?.start !== null && saved?.start !== undefined && saved.end !== null) input.setSelectionRange(saved.start, saved.end, saved.direction ?? undefined);
  }, [showText]);
  return { showText, revealable, props, disabled, concealLabel, revealLabel, fullWidth, inputRef, toggle };
}
```

## Content

```tsx
"use client";

import { InputGroup } from "../InputGroup";
import { Button } from "../../atoms/Button";
import { Input } from "../../atoms/Input";
import type { NativeSpecializedInputProps } from "../../../types/input-specializations";
import type { VisualProps, ComponentFullWidthProps, ComponentInvalidProps } from "../../../types/component-capabilities";
import { usePasswordInput } from "./usePasswordInput";

export interface PasswordInputProps extends NativeSpecializedInputProps, VisualProps, ComponentFullWidthProps, ComponentInvalidProps {
  revealLabel: string;
  concealLabel: string;
  revealable?: boolean;
}

export function PasswordInput(props: PasswordInputProps) {
  const { props: inputProps, inputRef, showText, revealable, disabled, fullWidth, concealLabel, revealLabel, toggle } = usePasswordInput(props);
  return (
    <InputGroup fullWidth={fullWidth} trailing={revealable ? <Button
      type="button"
      disabled={disabled}
      aria-label={showText ? concealLabel : revealLabel}
      aria-pressed={showText}
      onClick={toggle}
    >{showText ? concealLabel : revealLabel}</Button> : undefined}>
      <Input
        autoComplete="current-password"
        {...inputProps}
        ref={inputRef}
        disabled={disabled}
        fullWidth={fullWidth}
        type={showText ? "text" : "password"}
      />
    </InputGroup>
  );
}
```
