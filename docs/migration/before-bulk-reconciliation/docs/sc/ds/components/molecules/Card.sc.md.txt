## Hooks

```ts
"use client";

import { useId } from "react";
import type { CardProps } from "./Card";

export function useCard(input: CardProps) {
  const { children, className, elevation = "raised", shape, fullWidth = true, ...props } = input;
  const titleId = useId();
  return { children, className, props, titleId, elevation, shape, fullWidth };
}
```

## Content

```ts
"use client";

import { createContext, useContext } from "react";

export interface CardContextValue { titleId?: string; }
export const CardContext = createContext<CardContextValue | null>(null);

export function useCardContext() {
  const context = useContext(CardContext);
  if (!context) throw new Error("Card parts must be rendered inside Card.");
  return context;
}
```

```tsx
"use client";

import type { NativeSectionRootProps } from "../../../types/collection-contracts";
import type { ComponentChildrenProps } from "../../../types/compound-composition";
import { collectCompoundParts } from "../../../helpers/compound";
import { CardContext } from "../../../internal/card/CardContext";
import { CardHeader } from "../../atoms/CardHeader";
import { CardBody } from "../../atoms/CardBody";
import { CardFooter } from "../../atoms/CardFooter";
import { Surface } from "../../atoms/Surface";
import type { ComponentShapeProps, ComponentFullWidthProps } from "../../../types/component-capabilities";
import type { ComponentElevation } from "../../../types/layout";
import { useCard } from "./useCard";
import styles from "./Card.module.css";

export interface CardProps extends NativeSectionRootProps, ComponentChildrenProps, ComponentShapeProps, ComponentFullWidthProps {
  elevation?: ComponentElevation;
}

export function Card(props: CardProps) {
  const view = useCard(props);
  const parts = collectCompoundParts(view.children, { header: CardHeader, body: CardBody, footer: CardFooter }, ["body"]);
  return (
    <CardContext value={{ titleId: parts.header ? view.titleId : undefined }}>
      <Surface
        as="article"
        elevation={view.elevation}
        shape={view.shape}
        data-full-width={view.fullWidth || undefined}
        {...view.props}
        className={[styles.root, view.className].filter(Boolean).join(" ")}
        gap="0"
        aria-labelledby={parts.header ? view.titleId : undefined}
      >
        {parts.header}
        {parts.body}
        {parts.footer}
      </Surface>
    </CardContext>
  );
}
```

- [x] The elevation / fullWidth / shape contract uses raised / true / inherited.
- [x] The children contract uses Required Body; optional Header/Footer.
- [x] Native actions live inside Body/Footer, preserving independent keyboard semantics.
- [x] Use Card for self-contained article content; generic nonarticle panels use Surface.
- [x] Do not infer a clickable panel or wrap nested buttons in a single anchor.
- [x] No media loader, data fetch or business record shape.
- [x] useCard is colocated with Card and called at the top of render; state, effects, behavioral refs and handlers remain in that hook.

- [x] All three public parts provide their own internal spacing; the Card root has no additional inter-region gap or duplicate outer content padding.

- [x] The primary Showcase demonstrates complete text, icon/divider/image and avatar/profile compositions using actual Header, Body and Footer identities.

- [x] Elevation, shape and width examples are separated from the primary compositions rather than multiplying complete Cards into a Cartesian product.

- [x] Targeted browser checks verify padded regions, local image loading, decorative icons/avatar semantics, independent actions and no page-wide overflow at narrow widths.
