## Hooks

```tsx
"use client";

import { useId } from "react";
import { useChoiceGroupContext } from "../../../internal/choicegroup/ChoiceGroupContext";
import { useCheckboxGroupContext } from "../../../internal/checkboxgroup/CheckboxGroupContext";
import type { CheckboxGroupItemProps } from "./CheckboxGroupItem";

export function useCheckboxGroupItem({ label, value, disabled, className, ...props }: CheckboxGroupItemProps) {
  const group = useCheckboxGroupContext();
  const choice = useChoiceGroupContext();
  const inputId = useId();
  const describedBy = [...new Set([...(choice.describedBy?.split(/\s+/).filter(Boolean) ?? []), ...(props["aria-describedby"]?.split(/\s+/).filter(Boolean) ?? [])])].join(" ") || undefined;
  // The fieldset handles selection after native caller events have bubbled.
  function onChange() {}
  return { group, choice, props, label, value, disabled, inputId, describedBy, className, onChange };
}
```

## Content

```tsx
"use client";

import type { LabelHTMLAttributes } from "react";
import { Checkbox } from "../../atoms/Checkbox";
import { useCheckboxGroupItem } from "./useCheckboxGroupItem";
import styles from "./CheckboxGroupItem.module.css";

export interface CheckboxGroupItemProps extends LabelHTMLAttributes<HTMLLabelElement> {
  value: string;
  label: string;
  disabled?: boolean;
}

export function CheckboxGroupItem(input: CheckboxGroupItemProps) {
  const { group, choice, props, label, value, disabled, inputId, describedBy, className, onChange } = useCheckboxGroupItem(input);
  return (
    <div className={styles.root}>
      <Checkbox
        id={inputId}
        name={group.name}
        value={value}
        checked={group.isChecked(value)}
        disabled={group.disabled || disabled}
        invalid={group.invalid ?? choice.invalid}
        aria-describedby={describedBy}
        data-choice-group={group.id}
        onChange={onChange}
      />
      <label
        {...props}
        htmlFor={inputId}
        className={className}
      >{label}</label>
    </div>
  );
}
```
