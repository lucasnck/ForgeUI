## Hooks

```tsx
"use client";

import { useImperativeHandle } from "react";
import type { KeyboardEvent } from "react";
import { useMaskedInput } from "../../../hooks/mask/useMaskedInput";
import type { MaskedInputProps } from "./MaskedInput";

export function useMaskedInputView({ mask, value, onValueChange, onKeyDown, ref, ...props }: MaskedInputProps) {
  const masked = useMaskedInput({ mask, value, onValueChange });
  const { inputRef } = masked;
  useImperativeHandle(ref, () => inputRef.current!, [inputRef]);
  function handleKeyDown(event: KeyboardEvent<HTMLInputElement>) {
    onKeyDown?.(event);
    if (!event.defaultPrevented && !props.disabled && !props.readOnly) masked.onKeyDown(event);
  }
  return { ...masked, props, handleKeyDown };
}
```

## Content

```tsx
"use client";

import { Input } from "../../atoms/Input";
import type { NativeMaskedInputProps } from "../../../types/native-control-contracts";
import type { VisualProps, ComponentFullWidthProps, ComponentInvalidProps } from "../../../types/component-capabilities";
import { useMaskedInputView } from "./useMaskedInputView";

export interface MaskedInputProps extends NativeMaskedInputProps, VisualProps, ComponentFullWidthProps, ComponentInvalidProps {
  value: string;
  onValueChange(value: string): void;
  mask: string;
}

export function MaskedInput(props: MaskedInputProps) {
  const { props: inputProps, inputRef, displayValue, inputMode, maxLength, onChange, handleKeyDown } = useMaskedInputView(props);
  return (
    <Input
      {...inputProps}
      ref={inputRef}
      type="text"
      value={displayValue}
      inputMode={inputMode}
      maxLength={maxLength}
      onChange={onChange}
      onKeyDown={handleKeyDown}
    />
  );
}
```

## Criteria

- [x] Given a raw value, the rendered control shows the mask applied and the literals appear only after the preceding position is filled.
- [x] Given typing, `onValueChange` receives characters only, without literals, and `value` always round-trips through `applyMask`.
- [x] Given a paste that already contains literals, the resulting raw value matches the unmasked paste.
- [x] Given backspace over a separator, the separator is not deleted alone and the caret lands on the previous position.
- [x] Given a numeric-only mask, the control renders `inputmode="numeric"`; given a mask with letters, it renders `inputmode="text"`.
- [x] Given the mask length, `maxlength` equals it and no longer value can be entered.
- [x] Given `invalid`, `aria-invalid` reflects it; given composition with FormField, `id` and `aria-describedby` reach the control.
- [x] Given the component source, no masking library is imported and no mask literal of its own is declared.
- [x] Given the delivery, `npm run lint`, `npm run test:contracts`, `npm run test:tokens` and `npm run test:design-system` pass.
- [x] Includes the raw-value channel, the accessibility attributes the mask dictates (`inputMode`, `maxLength`) and the visual capabilities of the Input atom.
- [x] Excludes mask semantics (a mask is structural: this component does not know that a tax id has a check digit or that a date has a calendar), calendar pickers, locale formatting, validation messaging and every domain mask.
