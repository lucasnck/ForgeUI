## Hooks

```tsx
"use client";

import { useRef } from "react";
import { usePresence } from "../../../hooks/motion";
import type { PopupListboxProps } from "./PopupListbox";

export function usePopupListboxView<Item>({ popup, items, getOptionLabel, getOptionKey, listboxLabel, animated = true, animationEffect = "scale" }: PopupListboxProps<Item>) {
  const elementRef = useRef<HTMLUListElement>(null);
  const open = popup.isOpen && items.length > 0;
  const { present, phase, onTransitionEnd } = usePresence({ open, elementRef, enabled: animated && animationEffect !== "none" });
  return { popup, items, getOptionLabel, getOptionKey, listboxLabel, animated, animationEffect, elementRef, open, present, phase, onTransitionEnd };
}
```

## Content

```tsx
"use client";

import type { ComponentAnimationEffectProps } from "../../../types/animation";
import type { ComponentPopupProps, PopupListboxContract } from "../../../types/popup";
import { PopupListboxItem } from "../../atoms/PopupListboxItem";
import { usePopupListboxView } from "./usePopupListboxView";
import effects from "../../../styles/animations/surface-animations.module.css";
import styles from "./PopupListbox.module.css";

export interface PopupListboxProps<Item> extends ComponentPopupProps, ComponentAnimationEffectProps {
  popup: PopupListboxContract;
  items: readonly Item[];
  getOptionLabel(item: Item): string;
  getOptionKey?(item: Item, index: number): string;
}

export function PopupListbox<Item>(input: PopupListboxProps<Item>) {
  const { popup, items, getOptionLabel, getOptionKey, listboxLabel, animationEffect, animated, elementRef, open, present, phase, onTransitionEnd } = usePopupListboxView(input);
  if (!present) return null;
  return (
    <ul {...popup.listboxAttributes} ref={elementRef} className={[styles.list, effects[animationEffect]].join(" ")} aria-label={listboxLabel} aria-hidden={!open || undefined} inert={!open || undefined} data-animated={animated} data-motion-state={phase} data-motion-focus={popup.activeIndex >= 0 || undefined} onTransitionEnd={onTransitionEnd}>
      {items.map((item, index) => (
        <PopupListboxItem key={getOptionKey?.(item, index) ?? popup.optionId(index)} popup={popup} index={index} label={getOptionLabel(item)} animated={animated} />
      ))}
    </ul>
  );
}
```

## Tests

- V2 Public popup adapter: verify the specified actions/outcome, reset and agreement with its minimal copyable code.

- An initially closed or empty list renders no surface; a later exit retains only inert aria-hidden content.
- Each pointer-handler factory captures its own option index and accepts once.
- Closed retained options cannot request selection.
- animationEffect=none and animated=false settle without transition events.
- Virtual active-option relationships update immediately without cancelling the entry transition.

## Criteria

- [x] Given a closed contract, PopupListbox renders nothing; given an open contract with an empty `items`, it also renders nothing.
- [x] Given an open contract and three items, exactly one `ul[id][role=listbox]` and three `li[role=option]` elements render, in the caller's order.
- [x] Given an active index, only that option has `aria-selected="true"` and `data-active`, and the controlling input contract's `aria-activedescendant` resolves to that option id.
- [x] Given `listboxLabel`, the list exposes it as its accessible name; given no label, the list renders without an `aria-label` attribute.
- [x] Given rendered option text, it is exactly `getOptionLabel(item)`, and no matching, filtering or value formatting happens inside the component.
- [x] Given no `getOptionKey`, keys equal `popup.optionId(index)`; given `getOptionKey`, the returned value is used.
- [x] Given a pointer press on an option, `pointerdown` default is prevented and `popup.acceptOption(index)` runs once for that index, with no handler on `click`.
- [x] Given two items with identical labels, both options render with distinct ids.
- [x] Given the module CSS, it declares only `composes` rules and no token, numeric or color value.
- [x] Given the delivery, `npm run lint`, `npm run typecheck:design-system`, `npm run test:contracts`, `npm run test:tokens` and `npm run test:design-system` pass.
- [x] Includes the open/closed decision, the listbox surface, one option element per item, option labelling, keys, the active-option marking and pointer acceptance through the contract's `acceptOption`.
- [x] Excludes disclosure and active-option state, keyboard handling, matching, filtering, ordering, value resolution and completion (owned by [Popup contracts](../../hooks/tech-popup-contracts.sc.md) and each control), rich or multi-line option content, icons, grouping, disabled options, multi-selection, scrolling and virtualization, and every business, API, session or locale concern.
- [x] Props are declared as a generic interface extending `ComponentPopupProps` and `ComponentAnimationEffectProps`; `listboxLabel` is inherited and never redeclared, and the ARIA attribute bags stay owned by `Popup contracts`.
- [x] The popup contract uses Required; Instance returned by `usePopupListbox` in the current render of the consuming control; never a cached or re-created object.
- [x] The items contract uses Required; Same length as the contract's `count`; rendering stops when it is empty.
