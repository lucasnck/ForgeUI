## Content

```tsx
import type { HTMLAttributes } from "react";
import type { ComponentAlignProps, ComponentElevation, ComponentShapeProps, ComponentSpaceProps, LayoutElement, } from "../../../types";
import styles from "./Surface.module.css";
export interface SurfaceProps extends HTMLAttributes<HTMLElement>, ComponentSpaceProps, ComponentAlignProps, ComponentShapeProps {
  as?: LayoutElement;
  elevation?: ComponentElevation;
}
export function Surface({ as = "div", elevation = "raised", gap, padding, align, shape, className, ...props }: SurfaceProps) {
  const Element = as;
  return (<Element {...props} className={[styles.root, className].filter(Boolean).join(" ")} data-elevation={elevation === "flat" ? undefined : elevation} data-gap={gap} data-padding={padding} data-align={align} data-shape={shape}/>);
}
```

- [x] Given `elevation="flat"`, no `data-elevation` attribute is rendered and the shadow resolves to none.
- [x] Given `elevation="floating"`, the recipe resolves `--shadow-floating`.
- [x] Given `padding` or `gap`, the matching attribute is rendered and the recipe resolves the `--space-*` token.
- [x] Given `shape`, `data-shape` is rendered and the radius follows the shared shape recipe.
- [x] Given the four themes and forced colors, the panel keeps a distinguishable boundary against the page.
- [x] Given a `className`, it is appended without replacing the recipe classes, and no role or tabindex is added.
- [x] Given the module CSS, only `composes` rules appear.
- [x] Given the delivery, `npm run lint`, `npm run test:contracts`, `npm run test:tokens` and `npm run test:design-system` pass.
