## Hooks

```tsx
"use client";

import { useImperativeHandle, useLayoutEffect, useRef } from "react";
import { useStepperContext } from "../../../internal/stepper/StepperContext";
import type { StepperHeaderProps } from "./StepperHeader";

export function useStepperHeader({ children, className, ref, ...props }: StepperHeaderProps) {
  const { label, registerNavigation } = useStepperContext();
  const navigationRef = useRef<HTMLElement>(null);
  useImperativeHandle(ref, () => navigationRef.current!, []);
  useLayoutEffect(() => registerNavigation(navigationRef), [registerNavigation]);
  return { props, children, className, label, navigationRef };
}
```

## Content

```tsx
"use client";

import type { ComponentPropsWithRef } from "react";
import { collectStepperChildren } from "../../../internal/stepper/collectStepperChildren";
import { StepperItem } from "../../molecules/StepperItem";
import { useStepperHeader } from "./useStepperHeader";
import styles from "./StepperHeader.module.css";

export interface StepperHeaderProps extends ComponentPropsWithRef<"nav"> {}

export function StepperHeader(input: StepperHeaderProps) {
  const { props, children, className, label, navigationRef } = useStepperHeader(input);
  collectStepperChildren(children, StepperItem, "StepperHeader");
  return (
    <nav {...props} role={undefined} className={[styles.root, className].filter(Boolean).join(" ")}
      aria-label={label}
      ref={navigationRef}
      tabIndex={-1}>
      <ol className={styles.list}>{children}</ol>
    </nav>
  );
}
```
