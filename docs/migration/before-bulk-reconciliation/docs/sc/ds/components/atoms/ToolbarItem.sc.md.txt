## Hooks

```ts
"use client";

import { useLayoutEffect, useImperativeHandle, useRef } from "react";
import type { FocusEventHandler } from "react";
import { useToolbarContext } from "../../../internal/toolbar/ToolbarContext";
import type { ToolbarItemProps } from "./ToolbarItem";

export function useToolbarItem({ itemKey, disabled, children, ref, onFocus, className, animated = true, ...props }: ToolbarItemProps) {
  const { registerItem, onItemFocus, tabIndex: getTabIndex } = useToolbarContext();
  const buttonRef = useRef<HTMLButtonElement>(null);
  useImperativeHandle(ref, () => buttonRef.current!, []);
  useLayoutEffect(() => {
    if (buttonRef.current?.querySelector("input, select, textarea, button, a[href], [role=slider], [role=radio], [role=toolbar], [tabindex]")) {
      throw new Error("ToolbarItem content must not contain nested interactive controls.");
    }
    return registerItem({ id: itemKey, ref: buttonRef, textValue: buttonRef.current?.textContent ?? "", disabled });
  }, [registerItem, itemKey, children, disabled]);
  const handleFocus: FocusEventHandler<HTMLButtonElement> = (event) => {
    onFocus?.(event);
    if (!event.defaultPrevented) onItemFocus(itemKey);
  };
  const tabIndex = getTabIndex(itemKey);
  return { tabIndex, buttonRef, disabled, children, className, animated, props, handleFocus };
}
```

## Content

```tsx
"use client";

import type { NativeActionButtonProps } from "../../../types/collection-contracts";
import type { ComponentAnimatedProps } from "../../../types/animation";
import { Button } from "../Button";
import { useToolbarItem } from "./useToolbarItem";
import styles from "./ToolbarItem.module.css";

export interface ToolbarItemProps extends NativeActionButtonProps, ComponentAnimatedProps {
  itemKey: string;
}

export function ToolbarItem(input: ToolbarItemProps) {
  const { tabIndex, buttonRef, disabled, children, className, animated, props, handleFocus } = useToolbarItem(input);
  return (
    <Button
      {...props}
      ref={buttonRef}
      type="button"
      disabled={disabled}
      tabIndex={tabIndex}
      onFocus={handleFocus}
      tone="neutral"
      appearance="ghost"
      data-animated={animated}
      className={[styles.root, className].filter(Boolean).join(" ")}
    >
      {children}
    </Button>
  );
}
```
