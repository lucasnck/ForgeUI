## Hooks

```ts
"use client";

import type { ComponentPropsWithRef } from "react";
import { usePaginationContext } from "../../../internal/pagination/PaginationContext";
import type { PaginationItemProps } from "./PaginationItem";

export function usePaginationItem({ label, page, disabled, onClick, children, className, animated = true, "aria-current": current, ...props }: PaginationItemProps) {
  const pagination = usePaginationContext();
  if (!label.trim()) throw new Error("PaginationItem requires a nonempty label.");
  if (!Number.isInteger(page) || page < 1 || page > pagination.pageCount) throw new Error("Invalid PaginationItem target.");
  const selected = pagination.page === page && current !== false && current !== "false";
  const unavailable = pagination.disabled || disabled;
  const handleClick: NonNullable<ComponentPropsWithRef<"button">["onClick"]> = (event) => {
    if (unavailable) return;
    onClick?.(event);
    if (!event.defaultPrevented) pagination.requestPage(page);
  };
  return { label, children, className, animated, props, selected, unavailable, handleClick };
}
```

## Content

```tsx
"use client";

import type { NativeCollectionButtonProps } from "../../../types/compound-composition";
import type { ComponentAnimatedProps } from "../../../types/animation";
import { Button } from "../Button";
import { usePaginationItem } from "./usePaginationItem";
import styles from "./PaginationItem.module.css";

export interface PaginationItemProps extends NativeCollectionButtonProps, ComponentAnimatedProps {
  page: number;
  label: string;
}

export function PaginationItem(props: PaginationItemProps) {
  const view = usePaginationItem(props);
  return (
    <Button
      {...view.props}
      type="button"
      aria-label={view.label}
      aria-current={view.selected ? "page" : undefined}
      disabled={view.unavailable}
      onClick={view.handleClick}
      tone="neutral"
      appearance={view.selected ? "soft" : "ghost"}
      data-animated={view.animated}
      className={[styles.root, view.className].filter(Boolean).join(" ")}
    >
      {view.children ?? props.page}
    </Button>
  );
}
```

- Server markup preserves native owner attributes, named semantic regions and permitted child hierarchy.
- Invalid root ranges/item targets, orphan Items and unsupported wrappers fail.
- Current-page/out-of-range/disabled requests are ignored; valid requests run once without changing controlled state.
- Caller onClick runs before requestPage; defaultPrevented cancels the request and disabled Items run neither handler.
- aria-current=false or "false" suppresses the current marker; absent aria-current derives from the root, and caller values cannot force a noncurrent target to become current.
