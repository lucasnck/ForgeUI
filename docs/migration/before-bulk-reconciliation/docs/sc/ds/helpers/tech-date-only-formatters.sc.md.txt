## Content

```ts
import type {Locale} from "@/lib/contracts/preferences";

export function normalizeDateOnly(value: string): string {
  if (!/^[0-9]{4}-[0-9]{2}-[0-9]{2}$/.test(value)) throw new RangeError("Invalid civil date");
  const [year, month, day] = value.split("-").map(Number);
  if (year < 1) throw new RangeError("Invalid civil date");
  const date = new Date(0);
  date.setUTCHours(0,0,0,0);
  date.setUTCFullYear(year, month-1, day);
  if (date.getUTCFullYear() !== year || date.getUTCMonth() !== month-1 || date.getUTCDate() !== day)
    throw new RangeError("Invalid civil date");
  return value;
}

export function formatDateOnly(value: string | null, locale: Locale,
  options: Pick<Intl.DateTimeFormatOptions, "dateStyle"> = {}): string {
  if (value === null) return "";
  const raw = normalizeDateOnly(value);
  const [year, month, day] = raw.split("-").map(Number);
  const date = new Date(0);
  date.setUTCHours(0,0,0,0);
  date.setUTCFullYear(year,month-1,day);
  return new Intl.DateTimeFormat(locale, {
    dateStyle: options.dateStyle ?? "short", timeZone: "UTC", calendar: "gregory",
  }).format(date);
}

export function isValidDateOnly(value: unknown): boolean {
  if (typeof value !== "string") return false;
  try { normalizeDateOnly(value); return true; } catch { return false; }
}
```

## Criteria

- [x] Canonical input is strict Gregorian YYYY-MM-DD with year 0001–9999; invalid values throw without Date rollover normalization. No localized date parsing is guessed.
- [x] UTC is an internal presentation anchor, not a user/business timezone conversion. Civil dates never shift by the user's offset or daylight-saving rules.
- [x] Existing infrastructure formatDate remains the instant/date-time formatter requiring an explicit timezone. This helper has no time-of-day, account timezone or server-date semantics.
- [x] Explicit locale/options produce deterministic SSR/client output. Null means absent; empty or malformed strings do not become epoch/today automatically.
- [x] No library, browser globals, current-clock access or provider dependency is introduced.

- [x] Preferred public consumption uses the typed formatters/validators namespaces. Parsing and normalization functions in this source owner are internal DS exports, omitted from the root public barrel. The namespace modules alias the same owning functions rather than copying algorithms.
