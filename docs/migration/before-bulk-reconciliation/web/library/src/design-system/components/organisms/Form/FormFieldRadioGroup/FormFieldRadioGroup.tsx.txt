"use client";
import type { FieldPathByValue, FieldValues } from "react-hook-form";
import type { FormFieldRegistrationProps } from "../../../../types";
import type { FormFieldOption } from "@/lib/forms";
import { FormField } from "../../../molecules/FormField";
import { Radio } from "../../../atoms/Radio";
import { Stack } from "../../../atoms/Stack";
import { useFormFieldRadioGroup } from "./useFormFieldRadioGroup";
export interface FormFieldRadioGroupProps<
  TValues extends FieldValues,
  TName extends FieldPathByValue<TValues, string> = FieldPathByValue<TValues, string>,
> extends FormFieldRegistrationProps<TValues, TName> {
  id?: string;
  disabled?: boolean;
  readOnly?: boolean;
  options: readonly FormFieldOption<string>[];
}

export function FormFieldRadioGroup<
  TValues extends FieldValues,
  TName extends FieldPathByValue<TValues, string> = FieldPathByValue<TValues, string>,
>(props: FormFieldRadioGroupProps<TValues, TName>) {
  const view = useFormFieldRadioGroup(props);

  return (
    <FormField
      controlId={view.controlId}
      label={props.label}
      hint={props.hint}
      error={view.message}
    >
      <>
        <div
          role="radiogroup"
          aria-label={props.label}
          aria-describedby={view.describedBy}
          aria-invalid={Boolean(view.message)}
          aria-required={props.required || undefined}
          aria-readonly={props.readOnly || undefined}
        >
          <Stack gap="2">
            {props.options.map((option, index) => (
              <label
                key={option.value}
                htmlFor={
                  index === view.firstEnabled ? view.controlId : view.controlId + "-" + index
                }
              >
                <Radio
                  id={index === view.firstEnabled ? view.controlId : view.controlId + "-" + index}
                  name={view.field.name}
                  ref={index === view.firstEnabled ? view.field.ref : undefined}
                  value={option.value}
                  checked={view.value === option.value}
                  disabled={view.field.disabled || option.disabled}
                  onChange={view.change}
                  onBlur={view.field.onBlur}
                  aria-describedby={view.describedBy}
                  invalid={Boolean(view.message)}
                />
                {option.label}
              </label>
            ))}
          </Stack>
        </div>
      </>
    </FormField>
  );
}
