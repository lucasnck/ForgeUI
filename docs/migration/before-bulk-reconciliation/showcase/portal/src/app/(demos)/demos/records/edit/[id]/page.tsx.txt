import { TemplateDemoForm, templateDemoIds } from "@/features/showcase-v2";

export function generateStaticParams() {
  return templateDemoIds.map((id) => ({ id }));
}

export default async function Page({ params }: { params: Promise<{ id: string }> }) {
  const { id } = await params;

  return <TemplateDemoForm id={id} />;
}
